        private void VitaDrawMap(SpriteBatch spriteBatch, Vector2 menuOrigin, float menuY, Color color)
        {
            // Same easing as DrawInventory: 0 = map inside the inventory, 1 = map view.
            var percentage = MathF.Sin(-MathF.PI / 2 + (_changeCount / ChangeTime) * MathF.PI) * 0.5f + 0.5f;

            const int margin = 8;
            var bigScale = _scale;
            while (_mapSize.Y * (bigScale + 1) <= Game1.WindowHeight - 2 * margin)
                bigScale++;
            var bigSize = _mapSize.Y * bigScale;
            var textboxTop = TextboxOverlay.DialogBoxTextBox.Y - 8 * _scale;
            var bigY = MathF.Max(margin, MathF.Min((Game1.WindowHeight - bigSize) / 2, textboxTop - bigSize - margin));

            var small = new Vector2(menuOrigin.X + _recMap.X, menuOrigin.Y + _recMap.Y);
            var big = new Vector2((int)((Game1.WindowWidth - bigSize) / 2), (int)(bigY - menuY));
            var position = Vector2.Lerp(small, big, percentage);
            var scale = MathHelper.Lerp(_scale, bigScale, percentage);
            var size = (int)(_mapSize.Y * scale);

            spriteBatch.End();
            _mapOverlay.VitaDrawScale = scale;
            _mapOverlay.Draw(spriteBatch, new Rectangle((int)position.X, (int)position.Y, size, size), color, Game1.GetMatrix);
            _mapOverlay.VitaDrawScale = 0;
            spriteBatch.Begin(SpriteSortMode.Deferred, null, SamplerState.PointClamp, null, null, null, Game1.GetMatrix);
        }
