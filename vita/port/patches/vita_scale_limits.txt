Vita: game scale limited to 1x-4x plus auto. Zooming out below 1x multiplies the visible world (and GPU work),
and the PC limit of 20x creates render targets the Vita cannot handle. Saved settings outside the range are
clamped on load, so a previously saved bad value cannot crash the game again.

FILE Game1.cs
FIND
            MaxGameScale = max_game_scale;
            PixelGridAlpha = pixel_grid_alpha;
            GameSettings.GameScale = max_game_scale + 1;
REPLACE
            MaxGameScale = Math.Min(max_game_scale, 4);
            PixelGridAlpha = pixel_grid_alpha;
            GameSettings.GameScale = MaxGameScale + 1;
END

FILE InGame/Pages/Settings/VideoSettingsPage.cs

                buttonWidth, sliderHeight, new Point(1, 2), -3, Game1.MaxGameScale + 1, 1, GameSettings.GameScale, 
REPLACE

END

END
