Vita: save files keep their previous version as "<file>.bak" (the background writer renames the old file instead of
deleting it). A save, slot or settings file that is missing or unreadable (nothing parsable in it) is loaded from
its backup instead, so a damaged file costs at most the progress since the save before it. Deleting a slot removes
the backups too, and copying a slot drops the target's backup, which belonged to the slot's previous contents.

        public bool LoadFile(string filePath)
        {
            VitaLoadedSaves++;
            if (VitaLoadParsed(filePath))
                return true;
            var backup = filePath + ".bak";
            if (!File.Exists(backup) || !VitaLoadParsed(backup))
            {
                Reset();
                return false;
            }
            System.Console.WriteLine("save: " + filePath + (File.Exists(filePath) ? " is unreadable" : " is missing") + ", loaded its backup");
            return true;
        }
