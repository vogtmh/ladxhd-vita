        // The world image while a menu pauses the game, and the target it is being drawn into this frame.
        private RenderTarget2D _vitaFrozenWorld;
        private RenderTarget2D _vitaWorldCapture;
        private Vector2 _vitaFrozenCamera;
        private float _vitaFrozenScale;
        private int _vitaFrozenFrames;

        public void DrawGame(SpriteBatch spriteBatch)
        {
            if (!Game1.ScreenDirect)
            {
                _vitaFrozenWorld = null;
                DrawGameRT(spriteBatch);
                return;
            }
            var device = Game1.Graphics.GraphicsDevice;

            // Settings changed in the menu (game scale, shadows, lights) show up through the camera check or the
            // refresh once a second.
            var snapshot = InGameOverlay.VitaWorldPaused && !MapManager.ObjLink.IsTransitioning ? Game1.MainRenderTarget : null;
            if (snapshot != null && snapshot == _vitaFrozenWorld && ++_vitaFrozenFrames < 60 &&
                _vitaFrozenCamera == MapManager.Camera.Location && _vitaFrozenScale == MapManager.Camera.Scale)
            {
                VitaDrawFrozenWorld(spriteBatch, snapshot);
                return;
            }
            _vitaFrozenWorld = null;
            _vitaWorldCapture = snapshot;
            _vitaFrozenCamera = MapManager.Camera.Location;
            _vitaFrozenScale = MapManager.Camera.Scale;
            _vitaFrozenFrames = 0;
END
