Vita: leaving a house rebuilt the whole overworld (about 17,000 objects, 0.8-1.6 s on the Vita, plus a collection
afterwards). The overworld now stays alive while the player is in a house, a cave, a dungeon or any other map that
is not itself large; it is parked outside the MapManager so dungeon rooms and side areas can be loaded meanwhile.
On return only the static terrain objects are kept (colliders, water, animated tiles, sprites,
fences, ... about 16,000); every other object of the map file is created again exactly as a fresh load would
(doors, spawners, NPCs, items, enemies), and so are cut bushes. Fence parts and stone spawners are kept too
(stone spawners are reset to their loaded state). Objects spawned at runtime are removed. A loaded
save, or something destroyed that cannot be traced back to a map entry, rebuilds the overworld as before.
The reset runs without a collection, so everything it allocates stays until the next one; it avoids temporary
tables over all objects (per-object VitaEntry/VitaStamp fields instead).

        private void VitaNoteSpawns(int index, int spawnedBefore)
        {
            for (var i = spawnedBefore; i < SpawnObjects.Count; i++)

        private static bool VitaGone(GameObject gameObject) => gameObject.IsDead || gameObject.Map == null;

        // UpdateDeleteObjects for thousands of objects: its List.Remove calls on the 17,000-object lists would dominate.
        private void VitaRemoveObjects(HashSet<GameObject> remove)
        {
            var removed = ++_vitaStamp;
            foreach (var gameObject in remove)
                gameObject.VitaStamp = removed;
            lock (_alwaysAnimateLock)
            {
                foreach (var gameObject in remove)
                    _alwaysAnimateSet.Remove(gameObject);
                // Objects that register and then die in their constructor (the egg owl) never reach the map lists, and
                // objects that left the map (Link's shadow, followers) register again when they come back.
                for (var i = AlwaysAnimateObjectsMain.Count - 1; i >= 0; i--)
                {
                    if (AlwaysAnimateObjectsMain[i] == null || AlwaysAnimateObjectsMain[i].IsDead || AlwaysAnimateObjectsMain[i].Map != Owner)
                    {
                        _alwaysAnimateSet.Remove(AlwaysAnimateObjectsMain[i]);
                        AlwaysAnimateObjectsMain.RemoveAt(i);
                    }
                }
            }
            VitaFilter(AlwaysAnimateObjectsMain, removed);
            VitaFilter(_freezePersistObjects, removed);
            VitaFilter(_poolSpawnedObjects, removed);

            foreach (var gameObject in remove)
            {
                gameObject.Map = null;
                if ((gameObject.ComponentsMask & DrawComponent.Mask) == DrawComponent.Mask)
                {
                    _drawPool.RemoveEntity(gameObject);
                    _drawPoolB.RemoveEntity(gameObject);
                }
                _gameObjectPool.RemoveEntity(gameObject);
            }

            // Rebuilt in object order, which is the order AddObjectToMap registered them in.
            _keyChangeListeners.Clear();
            foreach (var gameObject in _poolSpawnedObjects)
                if ((gameObject.ComponentsMask & KeyChangeListenerComponent.Mask) == KeyChangeListenerComponent.Mask)
                    _keyChangeListeners.Add((gameObject.Components[KeyChangeListenerComponent.Index] as KeyChangeListenerComponent).KeyChangeFunction);
        }

        private static void VitaFilter(List<GameObject> list, int removed)
        {
            var write = 0;
            for (var read = 0; read < list.Count; read++)
            {
                var gameObject = list[read];
                if (gameObject == null || gameObject.VitaStamp != removed)
                    list[write++] = gameObject;
            }
            list.RemoveRange(write, list.Count - write);
        }

        // False when an object that is gone cannot be traced back to a map entry that could create it again.
        public bool VitaCanReuse()
        {
            if (VitaInitialObjects == null || VitaItemObjects == null)
                return false;
            foreach (var gameObject in VitaInitialObjects)
            {
                if (VitaGone(gameObject) && gameObject.VitaEntry < 0)
                {
                    System.Console.WriteLine("map reuse: rebuilding, destroyed " + gameObject.GetType().Name);
                    return false;
                }
            }
            return true;
        }

        private bool VitaChildrenIntact(int index)
        {
            if (VitaChildren.TryGetValue(index, out var children))
                foreach (var child in children)
                    if (!VitaIsInitial(child) || !VitaIsStatic(child) || VitaGone(child))
                        return false;
            return true;
        }

        // Static entries like ObjFence only spawn their parts and mark themselves dead; nothing to redo while the
        // parts are still there.
        private bool VitaPlacedPartsIntact(int index, GameObject gameObject)
        {
            return gameObject.IsDead && !VitaIsInitial(gameObject) && VitaIsStatic(gameObject) && VitaChildrenIntact(index);
        }

        // A stone spawner's only state is visibility (its stones hide over dug holes); reset it like a fresh load.
        private bool VitaResetStoneSpawner(int index, GameObject gameObject)
        {
            if (VitaKindOf(gameObject) != 2)
                return false;
            var spawner = (ProjectZ.InGame.GameObjects.Things.ObjStoneSpawner)gameObject;
            if (!VitaIsInitial(spawner) || VitaGone(spawner) || !VitaChildrenIntact(index))
                return false;
            spawner.IsActive = true;
            spawner.Hidden = false;
            return true;
        }

        public void VitaResetForReuse()
        {
            var vitaClock = System.Diagnostics.Stopwatch.StartNew();
            SpawnObjects.Clear();
            DeleteObjects.Clear();

            // Entries that are not static terrain, or whose object (or a spawned object) is gone, are created again.
            var recreateSet = new HashSet<int>();
            for (var i = 0; i < VitaItemObjects.Count; i++)
            {
                var gameObject = VitaItemObjects[i];
                if (gameObject != null && (VitaPlacedPartsIntact(i, gameObject) || VitaResetStoneSpawner(i, gameObject)))
                    continue;
                if (gameObject == null || !VitaIsInitial(gameObject) || !VitaIsStatic(gameObject) || VitaGone(gameObject))
                    recreateSet.Add(i);
            }
            foreach (var gameObject in VitaInitialObjects)
                if (VitaGone(gameObject) && gameObject.VitaEntry >= 0)
                    recreateSet.Add(gameObject.VitaEntry);
            // In map file order, as LoadObjects creates them.
            var recreate = new List<int>(recreateSet);
            recreate.Sort();

            // Runtime spawns do not exist in a freshly loaded map; the entries above are replaced completely.
            var inPool = ++_vitaStamp;
            var remove = new HashSet<GameObject>();
            foreach (var gameObject in _poolSpawnedObjects)
            {
                gameObject.VitaStamp = inPool;
                if (!VitaIsInitial(gameObject) || gameObject.IsDead)
                    remove.Add(gameObject);
            }
            foreach (var i in recreate)
            {
                var item = VitaItemObjects[i];
                if (item != null && item.VitaStamp == inPool)
                    remove.Add(item);
                if (VitaChildren.TryGetValue(i, out var children))
                    foreach (var child in children)
                        if (child.VitaStamp == inPool)
                            remove.Add(child);
            }
            remove.Remove(MapManager.ObjLink);
            var tPlan = vitaClock.ElapsedTicks;
            VitaRemoveObjects(remove);
            var tRemove = vitaClock.ElapsedTicks;

            // Create them again in the same steps as LoadObjects, from the map state a fresh load starts with.
            _keyChanged = false;
            Owner.VitaResetFlags();
            _vitaCreating = true;
            var firstNew = _poolSpawnedObjects.Count;
            foreach (var i in recreate)
            {
                var isB = i >= ObjectList.Count;
                var entry = ProjectZ.InGame.SaveLoad.SaveLoadMap.VitaPristineItem(Owner.MapFileName, isB ? i - ObjectList.Count : i, isB);
                if (entry == null)
                    continue;
                if (VitaItemObjects[i] != null)
                    VitaRemoveInitial(VitaItemObjects[i]);
                if (VitaChildren.TryGetValue(i, out var oldChildren))
                {
                    foreach (var child in oldChildren)
                        VitaRemoveInitial(child);
                    VitaChildren.Remove(i);
                }
                var spawnedBefore = SpawnObjects.Count;
                var gameObject = GetGameObject(Owner, entry.Index, entry.Parameter);
                AddObjectToMap(gameObject, isB);
                VitaItemObjects[i] = gameObject;
                if (gameObject != null)
                    gameObject.VitaEntry = i;
                VitaNoteSpawns(i, spawnedBefore);
            }
            AddSpawnedObjects(true);
            UpdateKeyListeners();
            var tCreate = vitaClock.ElapsedTicks;
            for (var i = firstNew; i < _poolSpawnedObjects.Count; i++)
            {
                var spawnedBefore = SpawnObjects.Count;
                _poolSpawnedObjects[i].Init();
                VitaNoteChildren(_poolSpawnedObjects[i], spawnedBefore);
            }
            UpdateKeyListeners();
            AddSpawnedObjects();
            var tInit = vitaClock.ElapsedTicks;
            for (var i = firstNew; i < _poolSpawnedObjects.Count; i++)
                VitaAddInitial(_poolSpawnedObjects[i]);
            _vitaCreating = false;

            _resetPending = false;
            _holeResetPending = false;
            Owner.ResetHoleMap();
            TriggerKeyChange();
            var tEnd = vitaClock.ElapsedTicks;
            var msPerTick = 1000.0 / System.Diagnostics.Stopwatch.Frequency;
            System.Console.WriteLine("map reuse: timing plan " + (tPlan * msPerTick).ToString("0.0") + " remove " + ((tRemove - tPlan) * msPerTick).ToString("0.0") +
                " create " + ((tCreate - tRemove) * msPerTick).ToString("0.0") + " init " + ((tInit - tCreate) * msPerTick).ToString("0.0") +
                " rest " + ((tEnd - tInit) * msPerTick).ToString("0.0") + " ms");
            var poolEntries = _gameObjectPool.NoneTiledObjects.Count;
            foreach (var tile in _gameObjectPool.ComponentTiles)
                poolEntries += tile.GameObjects.Count;
            var drawEntries = _drawPool.NoneTiledObjects.Count + _drawPoolB.NoneTiledObjects.Count;
            foreach (var tile in _drawPool.ComponentTiles)
                drawEntries += tile.DrawComponents.Count;
            foreach (var tile in _drawPoolB.ComponentTiles)
                drawEntries += tile.DrawComponents.Count;
            System.Console.WriteLine("map reuse: removed " + remove.Count + ", recreated " + recreate.Count + " of " + VitaItemObjects.Count +
                " map entries | objects " + _poolSpawnedObjects.Count + ", pool entries " + poolEntries + ", draw entries " + drawEntries +
                ", always animate " + AlwaysAnimateObjectsMain.Count + ", key listeners " + _keyChangeListeners.Count);
        }

        // The overworld kept alive while the player is in a smaller map (house, cave, dungeon).
        private Map.Map _vitaKeptMap;
        private int _vitaKeptLoadCount;
        private int _vitaKeptSaveCount;
        private bool _vitaReuse;
        private bool _vitaJustReused;
        // Takes the kept map's place as MapManager.NextMap.
        private Map.Map _vitaSpareMap;

        // Desktop A/B: LADX_KEEP_DUNGEONS=0 reloads the overworld after dungeons as before.
        private static readonly bool VitaKeepInDungeons = System.Environment.GetEnvironmentVariable("LADX_KEEP_DUNGEONS") != "0";

        private static bool VitaShouldKeep(Map.Map previousMap, Map.Map newMap)
        {
            if (previousMap.Objects?.VitaInitialObjects == null || newMap.IsEgg || newMap.IsFinalMap)
                return false;
            if (!VitaKeepInDungeons && (newMap.IsDungeon || newMap.IsCastle))
                return false;
            return newMap.Objects.ObjectList.Count + newMap.Objects.ObjectListB.Count < ObjectManager.VitaKeepMinObjects;
        }

        private bool VitaCanReuseKeptMap(string mapFile)
        {
            var kept = _vitaKeptMap;
            if (kept == null || kept.VitaLoadCount != _vitaKeptLoadCount ||
                _vitaKeptSaveCount != SaveManager.VitaLoadedSaves || kept.MapName != mapFile)
                return false;
            return kept.Objects.VitaCanReuse();
        }

        // A loaded save replaces the world; the kept overworld would only hold memory.
        public void VitaReleaseKeptMap()
        {
            VitaDropKeptMap();
        }

        private void VitaDropKeptMap()
        {
            var kept = _vitaKeptMap;
            _vitaKeptMap = null;
            // Only release the objects if the map still holds what was kept and is not in use.
            if (kept == null || kept.VitaLoadCount != _vitaKeptLoadCount || kept == _gameMapManager.CurrentMap)
                return;
            kept.Objects.ReleaseObjects();
            kept.Objects = new ObjectManager(kept);
        }
