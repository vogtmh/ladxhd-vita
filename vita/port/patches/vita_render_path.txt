Vita render path. The PC renderer draws the world through up to three full-screen render targets, copies it into
the main target, copies that into the screen and blurs it five times for the UI. On the Vita GPU every target
switch loads and stores the whole surface (about 2 ms at 960x544), so that pipeline alone exceeded the frame.

Here every offscreen pass runs first and small (shadows and light map at a quarter resolution, the blurred map
edges as before), then the frame is drawn in one go straight to the screen: world, light multiplied on top,
fog, transitions and UI. The UI boxes use the game's own "blur disabled" look.

The PC pipeline (DrawGameRT) remains for what needs the finished world as a texture: the wobble transition,
the shock effect, the pixel grid and colour correction.

        private RenderTarget2D _lightMap;

        // dst * src: multiplies the finished world by the light map.
        private static readonly BlendState MultiplyBlend = new BlendState
        {
            ColorSourceBlend = Blend.DestinationColor,
            ColorDestinationBlend = Blend.Zero,
            AlphaSourceBlend = Blend.Zero,
            AlphaDestinationBlend = Blend.One
        };

        public bool CanDrawToScreen()
        {
            var main = Game1.MainRenderTarget;
            if (UseShockEffect || main == null || MapManager?.CurrentMap == null)
                return false;
            if (CurrentRenderWidth != main.Width || CurrentRenderHeight != main.Height)
                return false;
            return !((MapTransitionSystem)GameSystems[typeof(MapTransitionSystem)]).WobbleActive;
        }

        public void DrawGame(SpriteBatch spriteBatch)
        {
            if (!Game1.ScreenDirect)
            {
                DrawGameRT(spriteBatch);
                return;
            }
            var device = Game1.Graphics.GraphicsDevice;

            if (GameSettings.EnableShadows && MapManager.CurrentMap.UseShadows)
            {
                /// RT:CRASH BYPASS
                if (_shadowRenderTarget == null) return;

                // Same as the PC: rendering at half resolution and blurring keeps moving shadows from flickering.
                RenderShadows(spriteBatch);

                Resources.BlurEffectH.Parameters["pixelX"].SetValue(1.0f / _shadowRenderTarget.Width);
                Resources.BlurEffectV.Parameters["pixelY"].SetValue(1.0f / _shadowRenderTarget.Height);
                Resources.BlurEffectH.Parameters["mult0"].SetValue(0.35f);
                Resources.BlurEffectH.Parameters["mult1"].SetValue(0.15f);
                Resources.BlurEffectV.Parameters["mult0"].SetValue(0.35f);
                Resources.BlurEffectV.Parameters["mult1"].SetValue(0.15f);

                device.SetRenderTarget(_shadowRenderTargetBlur);
                device.Clear(Color.Transparent);
                spriteBatch.Begin(SpriteSortMode.Immediate, null, SamplerState.AnisotropicClamp, null, null, Resources.BlurEffectV, null);
                spriteBatch.Draw(_shadowRenderTarget, Vector2.Zero, Color.White);
                spriteBatch.End();

                device.SetRenderTarget(_shadowRenderTarget);
                device.Clear(Color.Transparent);
                spriteBatch.Begin(SpriteSortMode.Immediate, null, SamplerState.AnisotropicClamp, null, null, Resources.BlurEffectH, null);
                spriteBatch.Draw(_shadowRenderTargetBlur, Vector2.Zero, Color.White);
                spriteBatch.End();

                MapManager.CurrentMap.Objects.ShadowTexture = _shadowRenderTarget;
            }

            MapManager.PrepareBlur(spriteBatch);

            var useLight = MapManager.CurrentMap.UseLight;
            if (useLight)
            {
                var lightWidth = Math.Max(1, CurrentRenderWidth / 4);
                var lightHeight = Math.Max(1, CurrentRenderHeight / 4);
                if (_lightMap == null || _lightMap.Width != lightWidth || _lightMap.Height != lightHeight)
                {
                    _lightMap?.Dispose();
                    _lightMap = new RenderTarget2D(device, lightWidth, lightHeight);
                }
                device.SetRenderTarget(_lightMap);
                MapManager.DrawLight(spriteBatch, (float)lightWidth / CurrentRenderWidth);
                _lightRenderTarget = _lightMap;
            }

            device.SetRenderTarget(null);
            device.Clear(Color.Black);

            // draw the map
            MapManager.Draw(spriteBatch);

            var screen = new Rectangle(0, 0, CurrentRenderWidth, CurrentRenderHeight);
            if (useLight)
            {
                // Same as the light shader: lerp(world * light, white, lightState).
                var lightState = MapManager.CurrentMap.LightState;
                spriteBatch.Begin(SpriteSortMode.Deferred, MultiplyBlend, SamplerState.LinearClamp);
                spriteBatch.Draw(_lightMap, screen, Color.White * (1 - lightState));
                spriteBatch.End();

                if (lightState > 0)
                {
                    spriteBatch.Begin(SpriteSortMode.Deferred, BlendState.Additive, SamplerState.PointClamp);
                    spriteBatch.Draw(Resources.SprWhite, screen, new Color(lightState, lightState, lightState));
                    spriteBatch.End();
                }
            }

            // Draw the fog over the finished world render.
            MapDiscoveryFog.Draw(spriteBatch);

            foreach (var gameSystem in GameSystems)
                gameSystem.Value.Draw(spriteBatch);

            if (useLight && DrawPlayerOnTopPercentage > 0)
            {
                Resources.LightShader.Parameters["sprLight"].SetValue(_lightMap);
                Resources.LightShader.Parameters["lightState"].SetValue(DrawPlayerOnTopPercentage);
                Resources.LightShader.Parameters["mode"].SetValue(1);
                // The shader divides the screen position by these, and the light map is smaller than the screen.
                Resources.LightShader.Parameters["width"].SetValue(CurrentRenderWidth);
                Resources.LightShader.Parameters["height"].SetValue(CurrentRenderHeight);

                spriteBatch.Begin(SpriteSortMode.Deferred, null, SamplerState.PointClamp, null, null, Resources.LightShader, MapManager.Camera.TransformMatrix);
                Link.DrawTransition(spriteBatch);
                spriteBatch.End();
            }
            else if (DrawPlayerOnTopPercentage > 0)
            {
                spriteBatch.Begin(SpriteSortMode.Deferred, null, SamplerState.PointClamp, null, null, null, MapManager.Camera.TransformMatrix);
                Link.DrawTransition(spriteBatch);
                spriteBatch.End();
            }

            // debug stuff
            spriteBatch.Begin(SpriteSortMode.Deferred, null, SamplerState.PointClamp);
            MapManager.Camera.Draw(spriteBatch);
            spriteBatch.End();
        }
